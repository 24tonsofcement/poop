cmake_minimum_required(VERSION 3.22)
project(pulse_demucs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
file(GLOB SOURCES "${DEMUCS_SOURCE}/src/*.cpp")
add_executable(pulse_demucs main.cpp ${SOURCES})
target_include_directories(pulse_demucs PRIVATE "${DEMUCS_SOURCE}/src" "${DEMUCS_SOURCE}/vendor/eigen")
target_compile_options(pulse_demucs PRIVATE -O3 -DNDEBUG -ffp-contract=fast)
set_target_properties(pulse_demucs PROPERTIES OUTPUT_NAME libpulse_demucs.so)
